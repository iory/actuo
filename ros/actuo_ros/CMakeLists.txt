cmake_minimum_required(VERSION 3.16)
project(actuo_ros)

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(ament_uv_virtualenv REQUIRED)
# Pin the venv to the ROS interpreter. Without it uv picks the newest python on
# the machine, and --system-site-packages then hands that interpreter an
# ABI-mismatched numpy/rclpy built for the ROS one.
find_package(Python3 REQUIRED COMPONENTS Interpreter)

# actuo comes from the checkout this package is part of: two levels up.
get_filename_component(ACTUO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT EXISTS "${ACTUO_ROOT}/pyproject.toml")
  message(FATAL_ERROR
    "actuo_ros: expected the actuo repository root at ${ACTUO_ROOT} "
    "(ros/actuo_ros sits two levels below it), but it has no pyproject.toml")
endif()
configure_file(requirements.in.in "${CMAKE_CURRENT_BINARY_DIR}/requirements.in" @ONLY)
ament_generate_virtualenv(
  REQUIREMENTS "${CMAKE_CURRENT_BINARY_DIR}/requirements.in"
  PYTHON ${Python3_EXECUTABLE})

# actuo_ros/ (the nodes' modules and the ROS robot interface) on the
# workspace's PYTHONPATH, so a robot's own package can import it.
ament_python_install_package(${PROJECT_NAME})

ament_uv_install_python_scripts(
  PROGRAMS
    scripts/actuo_bus_node.py
    scripts/actuo_presence_markers.py
    scripts/actuo_watch_pose.py
  STRIP_PY_EXTENSION
)

# actuo's own commands (probe, emulator, calibration tools; [project.scripts]
# in the repository's pyproject.toml) as `ros2 run actuo_ros <command>`, run by
# this package's venv. Read from pyproject.toml so the two lists cannot drift.
file(STRINGS "${ACTUO_ROOT}/pyproject.toml" _actuo_scripts
  REGEX "^actuo-[a-z0-9-]+ = \"[a-z0-9_.]+:[a-z_]+\"$")
set(_actuo_cli_wrappers "")
foreach(_line IN LISTS _actuo_scripts)
  string(REGEX MATCH "^(actuo-[a-z0-9-]+) = \"([a-z0-9_.]+):([a-z_]+)\"$" _m "${_line}")
  set(_wrapper "${CMAKE_CURRENT_BINARY_DIR}/actuo_cli/${CMAKE_MATCH_1}.py")
  file(WRITE "${_wrapper}"
    "\"\"\"${CMAKE_MATCH_1}, from actuo (generated by actuo_ros' CMakeLists.txt).\"\"\"\n"
    "from ${CMAKE_MATCH_2} import ${CMAKE_MATCH_3}\n\n"
    "raise SystemExit(${CMAKE_MATCH_3}())\n")
  list(APPEND _actuo_cli_wrappers "${_wrapper}")
endforeach()
if(NOT _actuo_cli_wrappers)
  message(FATAL_ERROR "actuo_ros: found no actuo-* scripts in ${ACTUO_ROOT}/pyproject.toml")
endif()
ament_uv_install_python_scripts(PROGRAMS ${_actuo_cli_wrappers} STRIP_PY_EXTENSION)

install(DIRECTORY udev DESTINATION share/${PROJECT_NAME} USE_SOURCE_PERMISSIONS)
# Board data a robot config refers to, e.g.
#   symbols: package://actuo_ros/config/kondoh7_v0.6.5_symbols.yaml
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  ament_uv_check_virtualenv()
endif()

ament_package(CONFIG_EXTRAS cmake/actuo_ros-extras.cmake.in)
